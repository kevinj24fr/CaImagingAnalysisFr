Installation

# Install from GitHub
devtools::install_github("kevinj24fr/CaImagingAnalysisFr")

CaExperiment Object

Unified Seurat-style container for all calcium imaging data, analysis results, and provenance tracking.

Preprocessing

Motion correction, neuropil subtraction, dF/F calculation, and spike inference (OASIS, threshold, Bayesian).

Dimensionality Reduction

PCA, GPFA, dPCA, tensor decomposition, and neural ODE fitting for latent dynamics.

Network Analysis

Functional connectivity, graph metrics, community detection, and causal discovery.

State Space Models

Hidden Markov Models (HMM), Switching Linear Dynamical Systems (SLDS), and changepoint detection.

Advanced Methods

Topological data analysis (TDA), recurrence quantification, optimal transport, and conformal prediction.

Quick Start

library(CaImagingAnalysisFr)

# Create CaExperiment and run analysis pipeline
ca <- CaExperiment(traces, frame_rate = 30) |>
  RunDFF(method = "rolling", window_size = 300) |>
  RunSpikes(method = "threshold", threshold_sd = 2.5) |>
  RunPCA(n_components = 15) |>
  RunConnectivity(method = "correlation") |>
  RunGraphMetrics() |>
  RunAssemblies(method = "pca", n_assemblies = 5) |>
  RunTransients(threshold_sd = 2.0)

# Access results
dff_traces <- GetTraces(ca, assay = "dff")
spikes <- GetSpikes(ca)
pca <- GetReduction(ca, "pca")
conn <- GetGraph(ca, "connectivity_correlation")

# Advanced analysis
ca <- RunHMM(ca, n_states = 3)
ca <- RunSLDS(ca, n_states = 2, latent_dim = 3)
ca <- RunTDA(ca)
ca <- RunNeuralODE(ca)
